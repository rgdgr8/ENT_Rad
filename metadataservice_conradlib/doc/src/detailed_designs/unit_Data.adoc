This variable of type EndpointConfig saves the result of the registration endpoint.

- batchSize: 20
- batchDebounce: 0.5
- batchMaxWait: 2
- connectionKeepalive: null
- sessionEnabled: false
- sessionCookieName: null
- maxRegistrationAge: 86400L
- retryDelay: 30L,
- statusCodeContentExpiry: mapOf(),
- statusCodeRetryBehavior: mapOf(401 to RetryBehavior.EXPONENTIAL, 403 to RetryBehavior.EXPONENTIAL, 429 to RetryBehavior.EXPONENTIAL, 500 to RetryBehavior.EXPONENTIAL, 502 to RetryBehavior.EXPONENTIAL, 503 to RetryBehavior.EXPONENTIAL, 504 to RetryBehavior.EXPONENTIAL)
